### Create USER STORY Draft Issues

Use this PowerShell script to auto-create draft issues for every US-XX.YY item in this file, while generating story-specific acceptance criteria and tasks from each story focus.

```powershell
# Required inputs
$owner = "mnichols08"
$projectNumber = "PROJECT_NUMBER"
$planPath = "RESEARCH/github-project.md"

# Safety toggle: set $true to preview without creating items
$dryRun = $true

$epicRegex = '^####\s+EPIC\s+(\d+)\s+-\s+(.+)$'
$storyRegex = '^\s*\d+\.\s+(US-(\d{2})\.\d{2})\s+(As\s+.+)$'
$sentenceRegex = '^As\s+.+?,\s+I want\s+(.+?)\s+so that\s+(.+?)\.?$'

$weekMap = @{
	"01" = "1"
	"02" = "1-2"
	"03" = "3-4"
	"04" = "4-5"
	"05" = "5-7"
	"06" = "7-10"
	"07" = "10-12"
	"08" = "12-14"
	"09" = "15-16"
	"10" = "17-18"
}

$trackMap = @{
	"01" = "QA"
	"02" = "Content"
	"03" = "Design"
	"04" = "Design"
	"05" = "Content"
	"06" = "Frontend"
	"07" = "Media"
	"08" = "QA"
	"09" = "QA"
	"10" = "Launch"
}

$riskMap = @{
	"01" = "Low"
	"02" = "Medium"
	"03" = "High"
	"04" = "Low"
	"05" = "High"
	"06" = "High"
	"07" = "Medium"
	"08" = "Medium"
	"09" = "High"
	"10" = "High"
}

function Get-StoryPackage {
	param([string]$prefix, [string]$focus, [string]$benefit)

	$ac = @(
		"Story focus is implemented: $focus",
		"User-value outcome is validated: $benefit"
	)

	$tasks = @(
		"Implement change for focus area: $focus",
		"Verify benefit is observable: $benefit"
	)

	switch ($prefix) {
		"01" {
			$ac += "Project governance artifact exists and is approved"
			$ac += "Definition of Ready and Definition of Done checklists are published"
			$tasks += "Configure project fields and conventions"
			$tasks += "Publish weekly governance cadence"
		}
		"02" {
			$ac += "Copy reflects canonical business facts"
			$ac += "No conflicting legacy claims remain"
			$tasks += "Audit pages for factual conflicts"
			$tasks += "Apply and verify corrected statements"
		}
		"03" {
			$ac += "Customer journey flow is documented for this path"
			$ac += "Navigation and CTA placement support the journey"
			$tasks += "Update IA or wireframe for target flow"
			$tasks += "Validate CTA path on desktop and mobile"
		}
		"04" {
			$ac += "Brand style and icon treatment is consistent"
			$ac += "Accessibility and contrast checks pass"
			$tasks += "Implement required visual or icon update"
			$tasks += "Run visual QA against style rules"
		}
		"05" {
			$ac += "Content is sourced from approved materials"
			$ac += "Stakeholder copy review is completed"
			$tasks += "Draft and place approved content block"
			$tasks += "Run plain-language and factual check"
		}
		"06" {
			$ac += "UI behavior is updated in all target components"
			$ac += "Regression checks confirm no reintroduced claims"
			$tasks += "Refactor component(s) tied to this story"
			$tasks += "Execute responsive and regression testing"
		}
		"07" {
			$ac += "Media and testimonial artifacts are approved for publish"
			$ac += "Privacy checks are completed for visual assets"
			$tasks += "Prepare and optimize story media assets"
			$tasks += "Integrate and validate gallery or testimonial module"
		}
		"08" {
			$ac += "Technical quality metric relevant to this story is met"
			$ac += "Audit evidence is attached for verification"
			$tasks += "Implement SEO, accessibility, or performance improvement"
			$tasks += "Run audit tool and document result"
		}
		"09" {
			$ac += "UAT evidence is logged for this scenario"
			$ac += "Sign-off path is updated with result"
			$tasks += "Execute UAT script steps for this story"
			$tasks += "Capture stakeholder decision and notes"
		}
		"10" {
			$ac += "Launch or hypercare readiness check for this scope passes"
			$ac += "Post-launch monitoring path is confirmed"
			$tasks += "Execute deployment or smoke-check step"
			$tasks += "Track issue in hypercare board if needed"
		}
	}

	return @{ AC = $ac; Tasks = $tasks }
}

$lines = Get-Content $planPath
$currentEpic = ""
$stories = @()
$storyIds = @()

foreach ($line in $lines) {
	if ($line -match $epicRegex) {
		$currentEpic = "EPIC $($Matches[1]) - $($Matches[2])"
		continue
	}

	if ($line -match $storyRegex) {
		$storyId = $Matches[1]
		$prefix = $Matches[2]
		$storyText = $Matches[3]
		$focus = ""
		$benefit = ""

		if ($storyText -match $sentenceRegex) {
			$focus = $Matches[1].Trim()
			$benefit = $Matches[2].Trim()
		}
		else {
			$focus = $storyText
			$benefit = "stakeholder value"
		}

		$stories += @{
			Id = $storyId
			Prefix = $prefix
			Epic = $currentEpic
			Text = $storyText
			Focus = $focus
			Benefit = $benefit
		}
		$storyIds += $storyId
	}
}

$uniqueCount = ($storyIds | Sort-Object -Unique).Count
if ($uniqueCount -ne 60) {
	throw "Expected 60 unique story IDs, found $uniqueCount"
}

foreach ($story in $stories) {
	$meta = Get-StoryPackage -prefix $story.Prefix -focus $story.Focus -benefit $story.Benefit
	$acText = ($meta.AC | ForEach-Object { "- [ ] $_" }) -join [Environment]::NewLine
	$taskText = ($meta.Tasks | ForEach-Object { "- [ ] $_" }) -join [Environment]::NewLine
	$title = "$($story.Id) [$($story.Epic)]"

	$body = @"
Epic: $($story.Epic)
User Story: $($story.Text)
Focus: $($story.Focus)
Week Window: $($weekMap[$story.Prefix])
Track: $($trackMap[$story.Prefix])
Risk: $($riskMap[$story.Prefix])

Acceptance Criteria:
$acText

Tasks:
$taskText
"@

	if ($dryRun) {
		Write-Host "DRY RUN -> $title"
	}
	else {
		gh project item-create $projectNumber --owner $owner --title "$title" --body "$body"
	}
}
```
